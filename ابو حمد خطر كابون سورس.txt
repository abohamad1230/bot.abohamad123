# -------------------- تحميل وحفظ الأهداف --------------------
def load_targets():
    if os.path.exists(STORAGE_FILE):
        try:
            with open(STORAGE_FILE, 'r', encoding='utf-8') as f:
                data = json.load(f)
                if isinstance(data, dict):
                    return set(data.get('targets', [])), data.get('lists', {})
                else:
                    return set(data), {}
        except:
            return set(), {}
    return set(), {}

def save_targets():
    try:
        data = {
            'targets': list(active_targets),
            'lists': target_lists
        }
        with open(STORAGE_FILE, 'w', encoding='utf-8') as f:
            json.dump(data, f, ensure_ascii=False, indent=2)
    except:
        pass

def generate_message(user_id, word_list):
    history = sent_lines_history.get(user_id, set())

    for _ in range(30):
        selected = random.choices(word_list, k=WORD_COUNT)
        random.shuffle(selected)
        message = " ".join(selected)

        if message not in history:
            if user_id not in sent_lines_history:
                sent_lines_history[user_id] = set()
            sent_lines_history[user_id].add(message)

            if len(sent_lines_history[user_id]) > 100:
                sent_lines_history[user_id] = set(list(sent_lines_history[user_id])[-50:])

            return message

    sent_lines_history[user_id] = set()
    selected = random.choices(word_list, k=WORD_COUNT)
    random.shuffle(selected)
    return " ".join(selected)

async def get_user_from_text(text):
    parts = text.split()
    if len(parts) < 2:
        return None, None

    identifier = parts[1]

    if identifier.isdigit():
        try:
            user = await client.get_entity(int(identifier))
            return user.id, str(identifier)
        except:
            return None, None

    try:
        username = identifier.replace('@', '').replace('t.me/', '').replace('https://t.me/', '')
        user = await client.get_entity(username)
        return user.id, f"@{username}"
    except:
        return None, None

async def get_user_from_reply(event):
    try:
        replied = await event.get_reply_message()
        if replied and replied.sender_id:
            user = await client.get_entity(replied.sender_id)
            username = f"@{user.username}" if user.username else f"ايدي {user.id}"
            return replied.sender_id, username, replied.chat_id, replied.id
    except:
        pass
    return None, None, None, None

async def spam_task(target_id, chat_id, reply_to, display_name, word_list, speed, list_num):
    if target_id in active_tasks:
        return

    print(f"🚀 بدء مهمة السبام على {target_id} (قائمة {list_num})")
    active_tasks[target_id] = True
    target_lists[str(target_id)] = list_num
    counter = 0

    try:
        while active_tasks.get(target_id):
            counter += 1
            message = generate_message(target_id, word_list)

            await show_typing(chat_id)

            await client.send_message(
                chat_id,
                message,
                reply_to=reply_to
            )
            print(f"📤 [{counter}] قائمة {list_num}: {message[:30]}...")

            await asyncio.sleep(speed)

    except Exception as e:
        print(f"❌ خطأ في السبام: {e}")
    finally:
        print(f"🛑 توقف السبام على {target_id}")
        active_tasks.pop(target_id, None)
        if str(target_id) in target_lists:
            del target_lists[str(target_id)]

async def nick_am_task(target_id, chat_id, name=None, reply_to=None):
    """ترسل سطور نيك ام - بدون اسم أو مع اسم"""
    if target_id in nick_am_tasks:
        return
    
    nick_am_tasks[target_id] = {
        'running': True,
        'chat_id': chat_id,
        'name': name,
        'reply_to': reply_to
    }
    
    print(f"🔥 بدء نيك ام على {target_id} (اسم: {name if name else 'بدون اسم'})")
    counter = 0
    
    try:
        while nick_am_tasks.get(target_id, {}).get('running', False):
            # اختيار سطر عشوائي
            line = random.choice(NICK_AM_LINES)
            
            # إذا كان فيه {name} والاسم موجود
            if '{name}' in line and name:
                message = line.format(name=name)
            else:
                # إذا ما فيه {name} أو ما في اسم
                message = line.replace('{name}', '').strip()
            
            # إرسال السطر
            await show_typing(chat_id)
            await client.send_message(chat_id, message, reply_to=reply_to)
            
            counter += 1
            print(f"📤 [نيك ام {counter}] {message[:30]}...")
            
            # استخدام السرعة العامة
            await asyncio.sleep(SPEED)
            
    except asyncio.CancelledError:
        print(f"🛑 تم إلغاء نيك ام على {target_id}")
    except Exception as e:
        print(f"❌ خطأ في نيك ام: {e}")
    finally:
        if target_id in nick_am_tasks:
            del nick_am_tasks[target_id]
        print(f"🛑 توقف نيك ام على {target_id}")

# -------------------- وظائف الكتم --------------------
async def mute_watcher(chat_id, user_id):
    if chat_id not in muted_tasks:
        muted_tasks[chat_id] = {}
    
    muted_tasks[chat_id][user_id] = True
    
    print(f"🔇 بدء مراقبة المكتم {user_id} في الشات {chat_id}")
    
    try:
        while muted_tasks.get(chat_id, {}).get(user_id, False):
            await asyncio.sleep(0.5)
    except Exception as e:
        print(f"❌ خطأ في مراقب المكتم: {e}")
    finally:
        if chat_id in muted_tasks and user_id in muted_tasks[chat_id]:
            del muted_tasks[chat_id][user_id]
        print(f"🔊 توقف مراقبة المكتم {user_id}")

# -------------------- وظيفة النقاط --------------------
async def dot_task(chat_id):
    try:
        while chat_id in dot_tasks and dot_tasks[chat_id]:
            await asyncio.sleep(60)
            await client.send_message(chat_id, ".")
    except asyncio.CancelledError:
        pass
    except Exception as e:
        print(f"❌ خطأ في النقاط: {e}")

# -------------------- وظيفة تفليش القروب --------------------
async def flush_group(chat_id):
    try:
        count = 0
        me = await client.get_me()
        async for user in client.iter_participants(chat_id):
            if not user.bot and user.id != me.id:
                try:
                    await client.ban_participant(chat_id, user.id)
                    count += 1
                    await asyncio.sleep(0.5)
                except:
                    pass
        
        return True, f"✅ تم طرد {count} عضو من القروب"
    except Exception as e:
        return False, f"❌ خطأ: {e}"

            await client.send_message('me', f"✅ تم حفظ {len(active_targets)} هدف")
            return

        if text == CMD_LOAD:
            loaded_targets, loaded_lists = load_targets()
            active_targets.update(loaded_targets)
            target_lists.update(loaded_lists)
            await client.send_message('me', f"✅ تم تحميل {len(loaded_targets)} هدف")
            return

        # ----- أوامر الكتم -----
        if text == 'كتم' and event.is_reply:
            try:
                replied = await event.get_reply_message()
                if replied and replied.sender_id:
                    target_id = replied.sender_id
                    chat_id = event.chat_id
                    
                    if chat_id not in muted_users:
                        muted_users[chat_id] = {}
                    muted_users[chat_id][str(target_id)] = True
                    save_muted()
                    
                    asyncio.create_task(mute_watcher(chat_id, target_id))
                    
                    try:
                        user = await client.get_entity(target_id)
                        name = f"@{user.username}" if user.username else user.first_name
                    except:
                        name = str(target_id)
                    
                    await client.send_message('me', f"🔇 تم كتم {name}")
                    
                    try:
                        await client.delete_messages(chat_id, event.id)
                    except:
                        pass
                else:
                    await client.send_message('me', f"❌ رد على رسالة الشخص")
            except Exception as e:
                await client.send_message('me', f"❌ خطأ: {e}")
            return

        if text == 'فك كتم' and event.is_reply:
            try:
                replied = await event.get_reply_message()
                if replied and replied.sender_id:
                    target_id = replied.sender_id
                    chat_id = event.chat_id
                    
                    if chat_id in muted_users and str(target_id) in muted_users[chat_id]:
                        del muted_users[chat_id][str(target_id)]
                        save_muted()
                        
                        if chat_id in muted_tasks and target_id in muted_tasks[chat_id]:
                            muted_tasks[chat_id][target_id] = False
                        
                        try:
                            user = await client.get_entity(target_id)
                            name = f"@{user.username}" if user.username else user.first_name
                        except:
                            name = str(target_id)
                        
                        await client.send_message('me', f"🔊 تم فك الكتم عن {name}")
                    else:
                        await client.send_message('me', f"❌ هذا المستخدم غير مكتم")
                else:
                    await client.send_message('me', f"❌ رد على رسالة الشخص")
            except Exception as e:
                await client.send_message('me', f"❌ خطأ: {e}")
            return

        if text == 'المكتمين':
            if not muted_users:
                await client.send_message('me', "📭 لا يوجد مكتمين")
                return
            
            result = "🔇 **قائمة المكتمين:**\n"
            for chat_id, users in muted_users.items():
                try:
                    chat = await client.get_entity(int(chat_id))
                    chat_name = chat.title if hasattr(chat, 'title') else str(chat_id)
                except:
                    chat_name = str(chat_id)
                
                result += f"\n📌 {chat_name}:\n"
                for user_id in users:
                    try:
                        user = await client.get_entity(int(user_id))
                        name = f"@{user.username}" if user.username else user.first_name
                    except:
                        name = str(user_id)
                    result += f"  • {name}\n"
            
            await client.send_message('me', result)
            return

        if text == 'كتم الكل':
            chat_id = event.chat_id
            
            await client.send_message('me', f"⚠️ **تحذير:** أنت على وشك كتم جميع الأعضاء في هذا الشات!\nاكتب `تأكيد` خلال 10 ثواني للتأكيد")
            
            @client.on(events.NewMessage(from_users='me'))
            async def confirm_mute_all(event):
                if event.out and event.raw_text == 'تأكيد':
                    try:
                        me = await client.get_me()
                        async for user in client.iter_participants(chat_id):
                            if not user.bot and user.id != me.id:
                                if chat_id not in muted_users:
                                    muted_users[chat_id] = {}
                                muted_users[chat_id][str(user.id)] = True
                                asyncio.create_task(mute_watcher(chat_id, user.id))
                        
                        save_muted()
                        await client.send_message('me', f"🔇 تم كتم جميع الأعضاء في هذا الشات")
                    except Exception as e:
                        await client.send_message('me', f"❌ خطأ: {e}")
                return
            
            await asyncio.sleep(10)
            return

        # ----- أمر إضافة قائمة -----
        if text.startswith('اضف '):
            new_text = text[4:].strip()
            if not new_text:
                await client.send_message('me', "❌ اكتب النص بعد الأمر، مثال: `اضف كس اختك يا ديوث`")
                return
            
            if new_text in CUSTOM_COMMANDS:
                await client.send_message('me', f"❌ الأمر `{new_text}` موجود بالفعل!")
                return
            
            if is_adding:
                await client.send_message('me', "⚠️ يوجد عملية إضافة قيد التنفيذ، اكتب `تم` لإنهائها أو `الغاء`")
                return
            
            is_adding = True
            add_target = new_text
            add_words = []
            
            await client.send_message('me', f"📝 اكتب الكلمات التي تريدها لقائمة `{new_text}` (كل كلمة في سطر منفصل، اكتب `تم` عندما تنتهي)")
            return

        # ----- معالجة كلمات الإضافة -----
        if is_adding and event.out:
            if text == 'تم':
                if len(add_words) > 0:
                    CUSTOM_COMMANDS[add_target] = add_words.copy()
                    save_custom_commands()
                    await client.send_message('me', f"✅ تم إضافة الأمر `{add_target}` مع {len(add_words)} كلمة!")
                else:
                    await client.send_message('me', f"❌ لم تضف أي كلمة، لم يتم حفظ القائمة")
                
                is_adding = False
                add_target = None
                add_words = []
                return
            
            elif text == 'الغاء':
                await client.send_message('me', f"❌ تم إلغاء إضافة الأمر `{add_target}`")
                is_adding = False
                add_target = None
                add_words = []
                return
            
            else:
                add_words.append(text.strip())
                await client.send_message('me', f"✅ تمت إضافة: `{text}`")
                return

        # ----- أمر عرض القوائم المخصصة -----
        if text == 'قوائم':
            if CUSTOM_COMMANDS:
                result = "📚 **القوائم المخصصة:**\n"
                for cmd, words in CUSTOM_COMMANDS.items():
                    result += f"• `{cmd}` - {len(words)} كلمة\n"
                await client.send_message('me', result)
            else:
                await client.send_message('me', "📭 لا توجد قوائم مخصصة")
            return

        # ----- أمر حذف قائمة -----
        if text.startswith('حذف قائمة '):
            cmd_to_delete = text[9:].strip()
            if cmd_to_delete in CUSTOM_COMMANDS:
                del CUSTOM_COMMANDS[cmd_to_delete]
                save_custom_commands()
                await client.send_message('me', f"✅ تم حذف القائمة `{cmd_to_delete}`")
            else:
                await client.send_message('me', f"❌ القائمة `{cmd_to_delete}` غير موجودة")
            return

        # ----- أمر شرح الإضافة -----
        if text == 'امراضافة':
            help_add = """
╔══════════════════════════════╗
║    إضافة أمر تشغيل جديد     ║
╚══════════════════════════════╝

1️⃣ اكتب: `اضف <اسم الأمر الجديد>`
   مثال: `اضف كس اختك يا ديوث`

2️⃣ اكتب الكلمات سطر سطر

3️⃣ اكتب `تم` عندما تنتهي ✅

4️⃣ اكتب `الغاء` للإلغاء ❌

📌 استخدم الأمر الجديد مع رد
   مثل: `كس اختك يا ديوث` (مع رد)

🗑️ للحذف: `حذف قائمة <اسم الأمر>`
📚 لعرض القوائم: `قوائم`
"""
            await client.send_message('me', help_add)
            return

        # ----- أمر إيقاف الكل -----
        if text == CMD_STOP_ALL:
            for tid in list(active_tasks.keys()):
                active_tasks[tid] = False
            active_tasks.clear()

            if chat_spam_active:
                chat_spam_active = False

            for chat_id in list(dot_tasks.keys()):
                dot_tasks[chat_id] = False
            dot_tasks.clear()

            for target_id in list(nick_am_tasks.keys()):
                nick_am_tasks[target_id]['running'] = False
            await asyncio.sleep(0.5)
            nick_am_tasks.clear()

            for target_id in list(basa_am_tasks.keys()):
                basa_am_tasks[target_id]['running'] = False
            await asyncio.sleep(0.5)
            basa_am_tasks.clear()

            await client.send_message('me', f"🛑 تم إيقاف الكل")
            return

        # ----- أوامر التشغيل المخصصة (مع رد) -----
        if text in CUSTOM_COMMANDS and event.is_reply:
            print(f"🔴 أمر مخصص مع رد: {text}")
            target_id, display_name, chat_id, reply_id = await get_user_from_reply(event)
            
            if target_id:
                active_targets.add(target_id)
                save_targets()
                
                asyncio.create_task(spam_task(
                    target_id, chat_id, reply_id, display_name,
                    CUSTOM_COMMANDS[text], SPEED, f"مخصص:{text}"
                ))
                
                await client.send_message('me', f"🚀 جالس ادعس بـ `{text}`")
            else:
                await client.send_message('me', f"❌ ما اشتغل")
            return

        # ----- أمر التشغيل مع رد (كس امختك - قائمة 1) -----
        if text == CMD_ACTIVATE_REPLY and event.is_reply:
            print("🔴 أمر مع رد - قائمة 1")
            target_id, display_name, chat_id, reply_id = await get_user_from_reply(event)
            
            if target_id:
                active_targets.add(target_id)
                save_targets()
                
                asyncio.create_task(spam_task(
                    target_id, chat_id, reply_id, display_name,
                    WORDS_REPLY, SPEED, 1
                ))
                
                await client.send_message('me', f"🚀 جالس ادعس امه هيا")
            else:
                await client.send_message('me', f"❌ ما اشتغل")
            return

        # ----- أمر السكس في الشات -----
        if text == CMD_CHAT_SPAM:
            print("🔴 أمر سكس في الشات")

            if chat_spam_active:
                chat_spam_active = False
                await client.send_message('me', f"وقفت")
            else:
                chat_spam_active = True
                chat_spam_chat_id = event.chat_id
                asyncio.create_task(chat_spam_task(event.chat_id, SPEED))
                await client.send_message('me', f"🚀 جالس ادعس ب الشات هيا")
            return

        # ----- أمر الإيقاف (واك) -----
        if text.startswith(CMD_STOP):
            print("🛑 أمر إيقاف")
            target_id, display_name = await get_user_from_text(text)

            if target_id:
                if target_id in active_tasks:
                    active_tasks[target_id] = False

                if target_id in active_targets:
                    active_targets.remove(target_id)
                    save_targets()

                await client.send_message('me', f"وقف")
            else:
                await client.send_message('me', ".")
            return
